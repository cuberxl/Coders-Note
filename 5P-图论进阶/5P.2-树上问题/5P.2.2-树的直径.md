#### $\S$ 树形 DP 求树的直径

- 适用场景：其可以在负权图上应用，但还原出具体路径较为繁琐。
- 状态表示：
  - $f[u]$ 表示经过 $u$ 的最长链长度，且 $u$ 为最浅节点（即不考虑父节点）。
  - $dis[u]$ 表示从 $u$ 走向子树的最远距离。
- 状态转移：
  - 设 $v_1, v_2$ 为 $u$ 的子节点，$d[v] = dis[v] + w[u \rightarrow v]$，则：
$$
f[u] = \max(d[v_1] + d[v_2])
$$
- 性质提取/优化：
  - 容易发现，当 $d[v_1], d[v_2]$ 分别为最大值和次大值，上式取得最大值。
  - 而我们可以发现，计算 $dis[u]$ 时，恰好需要遍历所有子节点，求出他们的 $d[son]$ 的最大值。
  - 所以，在计算  $dis[u]$ （最大值）的过程中，可以顺便维护出次大值，得出 $f[u]$。
- 特别的，当树上均为负权边，且树的直径允许只含一个点，我们可以只取一个节点作为直径，其长度为 $0$。
- 代码模板：

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 1e4 + 10, M = 4e4 + 10;
int idx, h[N], ne[M], to[M], w[M];
int dis[N], f[N];
void add(int a, int b, int c) {
    ne[idx] = h[a];
    to[idx] = b;
    w[idx] = c;
    h[a] = idx ++;
}
void dfs(int u, int from) {
    int mx = 0, smx = 0;
    for (int i = h[u]; ~i; i = ne[i]) {
        int v = to[i];
        if (v == from) continue;
        dfs(v, u);
        int d = dis[v] + w[i];
        if (d > mx)
            smx = mx, mx = d;
        else if (d > smx)
            smx = d;
    }
    dis[u] = mx;
    f[u] = mx + smx;
}
int main() {
    cin.tie(0) -> sync_with_stdio(0);
    int n; cin >> n;
    memset(h, -1, sizeof h);
    for (int i = 1; i < n; i ++) {
        int a, b, c; cin >> a >> b >> c;
        add(a, b, c); add(b, a, c);
    }
    dfs(1, 0);
    int ans = 0;
    for (int i = 1; i <= n; i ++) 
        ans = max(ans, f[i]);
    cout << ans;
    return 0;
}
```

#### $\S$ 两次搜索求树的直径

- 适用场景：容易还原出具体路径，但是只适用于非负权图，原因见下。
- tips：DFS、BFS均可运用于此算法，但考虑到栈空间问题，一般使用 BFS。
- 策略：
  - 从任意点出发，对树进行一次遍历，求出和出发点距离最远的节点，记作 $u$。
  - 从节点 $u$ 出发，再对树进行一次遍历，求出和 $u$ 距离最远的节点 $v$。
  - $u \rightarrow v$ 的路径即为树的直径。
- 证明 $u$ 一定是某条直径的端点：

  - 反证法，若 $u$ 不是某条直径的端点：
  
	  - 设出发点为 $a$，树的一条直径为 $p \rightarrow q$。
	  - 情况一：若 $a \rightarrow u$ 和 $p \rightarrow q$ 不相交
		  - 由树的连通性可知，两段路径存在一对点 $x, y$，满足 $x$ 在 $a \rightarrow u$ 上， $y$ 在 $p \rightarrow q$ 上，且 $x, y$ 连通。
		  - 设 $d_1 = x \rightarrow u, d_2 = x \rightarrow y， d_3 = y \rightarrow q$
		  - 由假设可知，$a \rightarrow x + d_1 \geq a \rightarrow x + d_2 + d_3$。
		  - 故 $d_1 \geq d_2 + d_3$。
		  - 故 $d_1 - d_2\geq d_3$。
		  - 故 $d_1 + d_2 \geq d_3$ （此步我们假设边权均为正）
		  - 故 $p \rightarrow u \geq p \rightarrow q$
		  - 故 $u$ 一定是某条直径的端点
		  
     -  情况二：若 $a \rightarrow u$ 和 $p \rightarrow q$ 相交
         -  设交点为 $x$，$d_1 = x \rightarrow u,d_2 = x \rightarrow q$.
         -  由假设可知，$a \rightarrow x + d_1 \geq a \rightarrow x + d_2$。
         - 故 $d_1 \geq d_2$。
         - 故 $p \rightarrow x + d_1 \geq p \rightarrow x + d _2$
         - 故 $u$ 一定是某条直径的端点。
   - 综上所述，$u$ 一定是某条直径的端点。
- 由上述证明可知，与 $u$ 距离最远的点一定是直径的另一端。
- 容易发现，当图不全为正权的时候，上述证明不成立，此时必须使用树形 DP 进行求解。 
- 代码模板：

```cpp
#include <bits/stdc++.h>

#define gc() getchar_unlocked()
#define pc(x) putchar_unlocked(x)
#define st first
#define nd second

using namespace std;
namespace IO {
    inline int rd() {
    	int x = 0, c = gc(), f = 1;
    	while (c != '-' && (c < '0'|| c > '9')) c = gc();
    	if (c == '-') f = -1, c = gc();
    	while ('0' <= c && c <= '9')
    		x = x * 10 + (c ^ 48), c = gc();
    	return x * f;
    }
    inline void wt(long long x) {
    	if (x < 0)
    		pc('-'), x = -x;
    	if (x > 9) wt(x / 10);
    	pc(x % 10 + '0');
    }
}
using namespace IO;
using PII = pair<int, int>;

const int N = 1e5 + 10, M = 2e5 + 10;

int idx, h[N], ne[M], to[M];
int dis[N];

void add(int a, int b) {
    ne[idx] = h[a];
    to[idx] = b;
    h[a] = idx ++;
}

PII bfs(int s) {
    memset(dis, -1, sizeof dis);
    queue<int> q;
    q.push(s); dis[s] = 0;
    int mx = s;
    while (q.size()) {
        int u = q.front(); q.pop();
        if (dis[u] > dis[mx])
            mx = u;
        for (int i = h[u]; ~i; i = ne[i]) {
            int v = to[i];
            if (dis[v] == -1) {
                dis[v] = dis[u] + 1;
                q.push(v);
            }
        }
    }
    return {mx, dis[mx]};
}

int main() {
    int n = rd();
    memset(h, -1, sizeof h);
    for (int i = 1; i < n; i ++) {
        int a = rd(), b = rd();
        add(a, b); add(b, a);
    }
    int u = bfs(1).st;
    int ans = bfs(u).nd;
    wt(ans);
    return 0;
}
```

- 保存具体路径：维护一个前驱数组，每次更新节点时记录前驱，最后从一个端点回溯即可。
-  代码模板：

```cpp
#include <bits/stdc++.h>

#define gc() getchar_unlocked()
#define pc(x) putchar_unlocked(x)
#define st first
#define nd second

using namespace std;
namespace IO {
    inline int rd() {
    	int x = 0, c = gc(), f = 1;
    	while (c != '-' && (c < '0'|| c > '9')) c = gc();
    	if (c == '-') f = -1, c = gc();
    	while ('0' <= c && c <= '9')
    		x = x * 10 + (c ^ 48), c = gc();
    	return x * f;
    }
    inline void wt(long long x) {
    	if (x < 0)
    		pc('-'), x = -x;
    	if (x > 9) wt(x / 10);
    	pc(x % 10 + '0');
    }
}
using namespace IO;
using PII = pair<int, int>;

const int N = 1e5 + 10, M = 2e5 + 10;

int idx, h[N], ne[M], to[M];
int dis[N], pre[N];
int tp, stk[N];

void add(int a, int b) {
    ne[idx] = h[a];
    to[idx] = b;
    h[a] = idx ++;
}

PII bfs(int s) {
    memset(dis, -1, sizeof dis);
    memset(pre, 0, sizeof pre);
    queue<int> q;
    q.push(s); dis[s] = 0;
    int mx = s;
    while (q.size()) {
        int u = q.front(); q.pop();
        if (dis[u] > dis[mx])
            mx = u;
        for (int i = h[u]; ~i; i = ne[i]) {
            int v = to[i];
            if (dis[v] == -1) {
                dis[v] = dis[u] + 1;
                pre[v] = u;
                q.push(v);
            }
        }
    }
    return {mx, dis[mx]};
}

int main() {
    int n = rd();
    memset(h, -1, sizeof h);
    for (int i = 1; i < n; i ++) {
        int a = rd(), b = rd();
        add(a, b); add(b, a);
    }
    int u = bfs(1).st;
    int v = bfs(u).st;
    while (v) {
        stk[++ tp] = v;
        v = pre[v];
    }
    for (int i = tp; i >= 1; i --)
        wt(stk[i]), pc(' ');
    return 0;
}
```
