#### 矩阵快速幂

- **概念**：利用快速幂的思想，计算出一个矩阵的幂。
- **代码模板：**
```cpp
using namespace std;
using lint = long long;
const int N = 1e2 + 10;
const int MOD = 1e9 + 7;
struct Mat {
    lint val[N][N];
    int n, m;
    Mat(int _n, int _m) {
        n = _n, m = _m;
        memset(val, 0, sizeof val);
    }
    Mat operator * (const Mat & b) const {
        Mat c(n, b.m);
        for (int i = 1; i <= n; i ++)
            for (int k = 1; k <= m; k ++) {
                if (val[i][k] == 0) continue;
                for (int j = 1; j <= b.m; j ++)
                    c.val[i][j] = mod(c.val[i][j] + val[i][k] * b.val[k][j]);
            }
        return c;
    }
};
Mat qpow(Mat a, lint k) {
    Mat ans(a.n, a.n);
    for (int i = 1; i <= a.n; i ++)
        ans.val[i][i] = 1;
    while (k) {
        if (k & 1) ans = ans * a;
        a = a * a;
        k >>= 1;
    }
    return ans;
}
int main() {
    cin.tie(0) -> sync_with_stdio(0);
    lint n, k; cin >> n >> k;
    Mat a(n, n);
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= n; j ++)
            cin >> a.val[i][j];
    Mat ans = qpow(a, k);
    for (int i = 1; i <= n; i ++) {
        for (int j = 1; j <= n; j ++)
            cout << ans.val[i][j] << ' ';
        cout << '\n';
    }
    return 0;
}
```
